**`src/components/layout/Topbar.jsx`**
```jsx
import { Bell, Search, Plus } from 'lucide-react';
import { user } from '../../data/mockData';
import './Topbar.css';

export default function Topbar() {
  const getGreeting = () => {
    const hour = new Date().getHours();
    if (hour < 12) return "Good morning";
    if (hour < 18) return "Good afternoon";
    return "Good evening";
  };

  const todayDate = new Intl.DateTimeFormat('en-US', { 
    weekday: 'long', month: 'short', day: 'numeric' 
  }).format(new Date());

  return (
    <div className="topbar">
      <div className="topbar-left">
        <h2 className="greeting">{getGreeting()}, {user.name}</h2>
        <span className="date">{todayDate}</span>
      </div>
      
      <div className="topbar-right">
        <button className="icon-btn">
          <Search size={18} />
        </button>
        <button className="icon-btn notif-btn">
          <Bell size={18} />
          <span className="notif-dot"></span>
        </button>
        <button className="btn-primary" onClick={() => {}}>
          <Plus size={16} />
          List a book
        </button>
      </div>
    </div>
  );
}
```

import { Link } from 'react-router-dom';
import StatCard from '../components/ui/StatCard';

            <Link to="/listed">Manage</Link>
          </div>
          {listedBooks.map(book => (

              key={book.id} 
              book={book} 
              rightContent={
                <>
                  <Badge status={book.status} text={book.status} />

                </>
              } 
            />
          ))}
        </div>
      </div>

      <div className="two-col">
        <div className="section">
          <div className="section-hd">
            <h3>Recent Activity</h3>

            <ActivityRow key={activity.id} activity={activity} />
          ))}
        </div>

        <div className="section">
          <div className="section-hd">
            <h3>Wishlist</h3>
            <Link to="/wishlist">View all</Link>

              key={book.id} 
              book={book}

                <Badge status={book.status === 'available' ? 'available' : 'waiting'} text={book.status} />
              } 
            />
          ))}
        </div>
      </div>
    </div>
  );
}
